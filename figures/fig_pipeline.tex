\node[bf, right=11mm of sam, yshift=12.5mm, text width=29mm, minimum height=14mm] (fuse) {\textbf{Relabel as Damper,}\\\textbf{Outlet or Switch}\\where the 2D module has $\geq$3 votes};

\draw[arr] (fuse) -- (cap);
\draw[arr] (cap) -- (out);

  \node[fit=(fuse)(cap), inner sep=3mm, fill=cfu!3, draw=cfu!40, dashed, rounded corners=3pt] (gf) {};

\node[lab, text=c2d] at (g2.north west) {2D small-component module (from the scanner's view)};
